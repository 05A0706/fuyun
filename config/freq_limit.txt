# ============================================================
# fuyun 频率限制配置 (安装时复制到 /sdcard/Android/yc/uperf/)
# 可随时编辑, 修改即时生效, 重启保留 (WebUI「频率限制」卡片可改)
#
# 作用: 限制 CPU 最高频率 (硬上限)。上限以下仍由 uperf 动态调频,
#       保留原有动态频率特性; 适合续航向/压制发热/息屏省电场景。
#
# 实现说明: 上限通过冻结 scaling_max_freq 节点生效, uperf 的
#       动态调频写入会被拦截, 无法突破上限; 解除后立即恢复动态。
# ============================================================

# 频率上限 (kHz): 0 = 动态不限制; 其他 = 上限, 如 1800000 = 1.8GHz
# 建议值: 800000 / 1000000 / 1200000 / 1400000 / 1600000 / 1800000 / 2000000
# 注: 上限高于某簇硬件上限时, 该簇自动保持不限制
FREQ_CAP=0

# 限制范围: big = 仅大核(最高频簇, 推荐); all = 全部核心
FREQ_SCOPE=big

# 息屏自动限频: 1 = 启用 (息屏时自动套用下方息屏上限, 亮屏恢复主上限)
# 0 = 关闭 (息屏维持主上限, 完全交给系统/墓碑类模块)
# 注: 息屏只压频率, 不干预应用 —— 息屏应用冻结/清理交给墓碑等专用模块
FREQ_OFFSCREEN=1

# 息屏时套用的上限 (kHz): 0 = 跟随主上限; 其他 = 息屏独立上限, 如 1200000 = 1.2GHz
FREQ_OFFSCREEN_CAP=1200000
