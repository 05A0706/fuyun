# ============================================================
# fuyun CPU 频率范围配置 (小/中/大核 min/max 可调)
# 安装时复制到 /sdcard/Android/yc/uperf/，可随时编辑，保存后立即生效。
#
# 值含义:
#   0       = 动态，不限制/不锁频
#   非 0    = kHz 数值，例如 1800000 = 1.8GHz（自动吸附到设备最近支持频点）
#   min=max = 锁频
#
# 频点以设备实测 scaling_available_frequencies 为准 (内置表兜底)，
# 非法值自动吸附: min 向上取、max 向下取；超出硬件范围忽略并记日志。
# 生效状态: WebUI 按 小/中/大核 显示 ✓/✗，日志前缀 "频率限制"。
# ============================================================

# 总开关: 1=启用本配置 0=停用 (停用后完全恢复动态)
FREQ_RANGE_ENABLE=0

# 小核 (8+ 为 CPU0-3, 8G2 为 CPU0-2)
LITTLE_MIN=0
LITTLE_MAX=0

# 中核 (8+ 为 CPU4-6, 8G2 为 CPU3-6)
MID_MIN=0
MID_MAX=0

# 大核 (CPU7)
BIG_MIN=0
BIG_MAX=0
