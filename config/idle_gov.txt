# ============================================================
# fuyun 辅助调速器配置 (安装时复制到 /sdcard/Android/yc/uperf/)
# 可随时编辑, 重启保留; 修改后最多一个轮询周期即生效, 无需重启
#
# 作用: 前台应用持续低 CPU 占用时 (如停留在静态页面), 压低 uperf
#       idle 场景的功率预算, 压制空闲频率省电; 一旦恢复交互立即还原。
#       触摸瞬间 uperf 自动切 touch 场景, 交互性能不受影响。
#
# 配套文件:
#   idle_whitelist.txt - 调速器白名单 (前台命中则永不压制)
# ============================================================

# 总开关: 1=启用 0=停用
IDLE_GOV=1

# 判定轮询间隔 (秒): 前台应用 CPU 采样频率
IDLE_INTERVAL=10

# 进入深度空闲需要的前台应用持续低 CPU 时长 (秒)
IDLE_TIMEOUT=30

# 前台应用 CPU 占用阈值 (占单核百分比): 低于该值视为空闲
# 视频解码/地图渲染等持续高占用应用会自动豁免, 无需进白名单
IDLE_CPU_THD=5

# 深度空闲时 uperf idle 场景的功率预算 (瓦): 越低频率压得越狠
# 参考: 0.8 ≈ 1GHz 出头, 0.5 更激进, 1.2 保守
# 建议先保守 (0.8-1.0) 实测不卡再往下调
IDLE_POWER_W=0.8
